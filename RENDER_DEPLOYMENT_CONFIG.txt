================================================================================
CIVICPULSE — RENDER DEPLOYMENT SPECIFICATION & CONFIGURATION TEXT FILE
================================================================================

This document contains all official Render configuration specifications,
blueprint YAML, build commands, and environment variables needed to deploy 
CivicPulse to Render (https://render.com).

--------------------------------------------------------------------------------
1. RENDER BLUEPRINT SPECIFICATION (render.yaml)
--------------------------------------------------------------------------------
Copy and paste this into your "render.yaml" file or use it as a Render Blueprint:

services:
  - type: web
    name: civicpulse-app
    runtime: python
    plan: free
    buildCommand: bash build.sh
    startCommand: uvicorn backend.app.main:app --host 0.0.0.0 --port $PORT
    healthCheckPath: /api/health
    autoDeploy: true
    envVars:
      - key: PYTHON_VERSION
        value: 3.11.9
      - key: ENVIRONMENT
        value: production
      - key: SECRET_KEY
        generateValue: true
      - key: DATABASE_URL
        value: sqlite:///./civicpulse.db

4. RENDER BUILD SCRIPT (build.sh)
--------------------------------------------------------------------------------
This shell script builds both the React Vite frontend and FastAPI Python backend:

#!/usr/bin/env bash
set -o errexit

if command -v python3 &>/dev/null; then
    PY_BIN=python3
else
    PY_BIN=python
fi

echo "1. Installing Python dependencies..."
$PY_BIN -m pip install --upgrade pip
$PY_BIN -m pip install -r requirements.txt

echo "2. Building React Frontend..."
cd frontend
npm install
npm run build
cd ..

echo "3. Initializing Database and ML Model..."
$PY_BIN -c "
from backend.app.database import engine, Base, SessionLocal
from backend.app.seed_data import seed_database
Base.metadata.create_all(bind=engine)
db = SessionLocal()
seed_database(db)
db.close()
print('CivicPulse initialization completed successfully.')
"

echo "=== Build Completed Successfully! ==="

6. DOCKERFILE OPTION (If deploying via Render Docker Runtime)
--------------------------------------------------------------------------------
FROM node:20-alpine AS frontend-builder
WORKDIR /app/frontend
COPY frontend/package*.json ./
RUN npm install
COPY frontend/ ./
RUN npm run build

FROM python:3.11-slim
WORKDIR /app
RUN apt-get update && apt-get install -y --no-install-recommends build-essential curl && rm -rf /var/lib/apt/lists/*
COPY requirements.txt ./
RUN pip install --no-cache-dir --upgrade pip && pip install --no-cache-dir -r requirements.txt
COPY backend/ ./backend/
COPY --from=frontend-builder /app/frontend/dist ./frontend/dist
ENV PORT=8000
ENV ENVIRONMENT=production
EXPOSE 8000
CMD ["sh", "-c", "uvicorn backend.app.main:app --host 0.0.0.0 --port ${PORT}"]
